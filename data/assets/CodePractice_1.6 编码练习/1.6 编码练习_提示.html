本节目标

  • 使用 ADAM 发送4017采集指令，并接受4017采集结果
  • 使用 ADAMExpansion 扩展类转换得到二氧化碳值


类库函数说明


public class ADAM : IDisposable

  {

    public ADAM();

    public ADAM(string portName);

    public ADAM(string portName, int baudRate);


    ~ADAM();


    public bool IsConnected { get; }


    public event EventHandler<ADAM4150EventArgs> ADAM4150DataReceived;

    public event EventHandler<byte[]> SourceSerialPortDataReceived;

    public event EventHandler<ADAM4017EventArgs> ADAM4017DataReceived;

    public event EventHandler<byte[]> RegularDataReceived;


    public void Close();

    public bool Connect(string portName, int baudRate);

    public bool Connect();

    public bool Connect(string portName);

    public void Dispose();

    public ADAM4017Data ReadADAM4017Data();

    public ADAM4150Data ReadADAM4150Data();

    public void Send(byte[] buffer);

    public void SendADAM4017Data();

    public void SendADAM4150Data();

    public void Switch(Switchs dig);

    protected virtual void Dispose(bool disposing);

  }


public static class ADAMExpansion

  {

    public static double AirPressure(this ADAM adam, int value);

    public static double AirQuality(this ADAM adam, int value);

    public static double CO2(this ADAM adam, int value);

    public static double Humidity(this ADAM adam, int value);

    public static double Illumination(this ADAM adam, int value);

    public static double O2(this ADAM adam, int value);

    public static double PM25(this ADAM adam, int value);

    public static double SoilHumidity(this ADAM adam, int value);

    public static double SoilTemperature(this ADAM adam, int value);

    public static double Temperature(this ADAM adam, int value);

    public static double WaterLevel(this ADAM adam, int value);

    public static double WaterTemperature(this ADAM adam, int value);

    public static double WindDirection(this ADAM adam, int value);

    public static double WindSpeed(this ADAM adam, int value);

  }
